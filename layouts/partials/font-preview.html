{{- /* Font preview widget — branch `fonts` only. Filesystem-driven font picker. Scans static/font/ at build time: every subfolder with a .ttf becomes an @font-face + a dropdown option. Drop a new font folder in → it appears. Selection applied to body inline (beats the single body font rule) and persisted in localStorage so it sticks across pages. */ -}} {{- $base := "static/font" -}} {{- $fonts := slice -}} {{- range $e := (os.ReadDir $base) -}} {{- if $e.IsDir -}} {{- $dir := printf "%s/%s" $base $e.Name -}} {{- $cands := slice -}} {{- range (os.ReadDir $dir) -}} {{- if strings.HasSuffix (lower .Name) ".ttf" -}}{{- $cands = $cands | append (printf "%s/%s" $dir .Name) -}}{{- end -}} {{- end -}} {{- $sub := printf "%s/static" $dir -}} {{- if (os.FileExists $sub) -}} {{- range (os.ReadDir $sub) -}} {{- if strings.HasSuffix (lower .Name) ".ttf" -}}{{- $cands = $cands | append (printf "%s/%s" $sub .Name) -}}{{- end -}} {{- end -}} {{- end -}} {{- if gt (len $cands) 0 -}} {{- $rep := index $cands 0 -}} {{- $found := false -}} {{- range $cands -}}{{- if and (not $found) (findRE `(?i)regular` .) (not (findRE `(?i)italic` .)) -}}{{- $rep = . -}}{{- $found = true -}}{{- end -}}{{- end -}} {{- $label := replaceRE `([a-z])([A-Z])` "$1 $2" (replace $e.Name "_" " ") -}} {{- $fonts = $fonts | append (dict "id" $e.Name "label" $label "url" (relURL (strings.TrimPrefix "static/" $rep))) -}} {{- end -}} {{- end -}} {{- end -}}
